# Real wx event delivery, without a GUI app, display, GL context or renderer.
if (UNIX)
    set(wxWidgets_USE_UNICODE ON)
    set(wxWidgets_USE_STATIC ${SLIC3R_STATIC})
endif ()
if (CMAKE_SYSTEM_NAME STREQUAL "Linux")
    set(wxWidgets_CONFIG_OPTIONS "--toolkit=gtk${SLIC3R_GTK}")
endif ()
find_package(wxWidgets 3.2 MODULE REQUIRED COMPONENTS base)
include(${wxWidgets_USE_FILE})
add_executable(gui_background_events_tests test_main.cpp test_background_events.cpp test_publication_events.cpp)
target_compile_features(gui_background_events_tests PRIVATE cxx_std_17)
target_include_directories(gui_background_events_tests PRIVATE ${CMAKE_SOURCE_DIR}/src ${CMAKE_SOURCE_DIR}/tests)
target_link_libraries(gui_background_events_tests PRIVATE Catch2::Catch2 TBB::tbb TBB::tbbmalloc libslic3r ${wxWidgets_LIBRARIES})
set_property(TARGET gui_background_events_tests PROPERTY FOLDER "tests")
add_test(NAME sla_background_events COMMAND gui_background_events_tests)
set_tests_properties(sla_background_events PROPERTIES LABELS "sla;gui;events;unit" TIMEOUT 60)
